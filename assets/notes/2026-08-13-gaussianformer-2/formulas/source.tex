\begin{aligned}
\alpha(\mathbf x;\mathbf G_i)
&=\exp\!\left[-\frac12(\mathbf x-\mathbf m_i)^{\mathsf T}
\mathbf\Sigma_i^{-1}(\mathbf x-\mathbf m_i)\right],\\
\alpha(\mathbf x)
&=1-\prod_{i=1}^{P}\left[1-\alpha(\mathbf x;\mathbf G_i)\right]
\qquad\text{(4--5)}
\end{aligned}
% END eq-01-probabilistic-union

% BEGIN eq-02-semantic-mixture
\begin{aligned}
\mathbf e(\mathbf x;\mathcal G)
&=\sum_{i=1}^{P}p(\mathbf G_i\mid\mathbf x)\,\widetilde{\mathbf c}_i\\[-0.35em]
&=\frac{\sum_{i=1}^{P}p(\mathbf x\mid\mathbf G_i)a_i\widetilde{\mathbf c}_i}
{\sum_{j=1}^{P}p(\mathbf x\mid\mathbf G_j)a_j}
\qquad\text{(6)}
\end{aligned}
% END eq-02-semantic-mixture
