% BEGIN eq-14-stcv
\begin{aligned}
\mathrm{STCV}_t
&=
\frac{
\sum \mathbf 1\!\left[
\begin{aligned}
&\left(\chi[\mathbf M_t\langle\mathbf P\rangle,T_t]\ne\mathbf P_t\right)\\[-2pt]
&{}\land
\left(\chi[\mathbf M_t\langle\mathbf P\rangle,T_t]\ne\mathrm{Free}\right)
\end{aligned}
\right]
}{
\sum \mathbf 1[\mathbf P_t\ne\mathrm{Free}]
}\\[-2pt]
&\qquad\text{(14)}
\end{aligned}
% END eq-14-stcv
