% Canonical, copyable formula source for the rendered PNGs in this directory.
% Paper formulas are transcribed from RoboBEV, TPAMI 2025.

% BEGIN eq-01-mce
\begin{aligned}
\mathrm{CE}_i
&=\frac{\sum_{l=1}^{3}(1-\mathrm{NDS})_{i,l}}
{\sum_{l=1}^{3}(1-\mathrm{NDS}^{\mathrm{baseline}}_{i,l})},\\
\mathrm{mCE}
&=\frac{1}{N}\sum_{i=1}^{N}\mathrm{CE}_i .
\qquad\text{(1)}
\end{aligned}
% END eq-01-mce

% BEGIN eq-02-mrr
\begin{aligned}
\mathrm{RR}_i
&=\frac{\sum_{l=1}^{3}\mathrm{NDS}_{i,l}}
{3\times\mathrm{NDS}_{\mathrm{clean}}},\\
\mathrm{mRR}
&=\frac{1}{N}\sum_{i=1}^{N}\mathrm{RR}_i .
\qquad\text{(2)}
\end{aligned}
% END eq-02-mrr
