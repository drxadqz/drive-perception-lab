[\mathbf F^{\mathrm{rgb}}_{\mathbf p},\mathbf F_{\mathbf p}]

\end{aligned}

\begin{bmatrix}\phi\\\omega\\r\end{bmatrix}
=\begin{bmatrix}
\operatorname{atan2}(y,x)\\
\arcsin(z/r)\\
\sqrt{x^2+y^2+z^2}
\end{bmatrix}
\qquad\text{(9)}
% END eq-02-spherical-projection
